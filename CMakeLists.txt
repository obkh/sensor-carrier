cmake_minimum_required(VERSION 3.16)
project(sensor_carrier C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_library(sht31 firmware/sht31.c)
target_include_directories(sht31 PUBLIC firmware)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(sht31 PRIVATE -Wall -Wextra -Werror -pedantic)
endif()
enable_testing()
add_executable(test_sht31 tests/test_sht31.c)
target_link_libraries(test_sht31 PRIVATE sht31)
add_test(NAME sht31 COMMAND test_sht31)
